\begin{frame}{项目动机}
  \begin{itemize}
    \item 具身智能系统常由感知、融合、控制等算子组成，并形成 DAG。
    \item 部分算子必须固定在机器人端侧设备，部分算子可以卸载到边侧主机。
    \item 真实调度不仅是图切分：
      \begin{itemize}
        \item 跨设备边会产生网络传输代价；
        \item 端侧和边侧都有有限执行队列；
        \item 端到端时延上限会直接排除不可行方案。
      \end{itemize}
    \item CoopInfer 用于快速编辑、求解、验证并可视化这类调度方案。
  \end{itemize}
\end{frame}

\begin{frame}{内层解析器：数据依赖与资源排队}
  对 DAG 按拓扑序遍历。对节点 $i$：
  \[
    DataReady_i =
    \max\left(
      release_i(f),
      \max_{j \in pred(i)}
      \left[
        F_j +
        \mathbb{I}(x_j \neq x_i)
        \left(L + \frac{s_{ji}}{B}\right)
      \right]
    \right)
  \]
  输入源节点可使用 $release_i(f)=phase_i+f\cdot period_i$ 表示第 $f$ 帧的生成时刻。
  源节点是零耗时的外部输入事件，不占用端侧或边侧计算队列；后续帧输入可以在前一帧下游计算尚未完成时继续按周期到达。
  \vspace{0.5em}
  数据就绪后，还要等待目标设备队列空闲：
  \[
    S_i =
    \begin{cases}
      \max(DataReady_i, time\_dev\_ready), & x_i=0 \\
      \max(DataReady_i, time\_host\_ready), & x_i=1
    \end{cases}
  \]
  \[
    F_i = S_i + (1-x_i)c_i^{dev} + x_i c_i^{host}
  \]
  端到端时延从第一个非输入计算或传输事件开始，到最后一个非输入输出节点完成结束。输入源释放会推迟下游工作，但不计入时延窗口内部。
\end{frame}

\begin{frame}{网络模型：串行传输与批处理}
  \begin{itemize}
    \item 所有跨设备传输共享一条网络工作队列：
      \[
        time\_network\_ready
      \]
    \item 传输必须等待源节点完成，并且等待网络队列空闲，因此不同传输不会重叠。
    \item 未开启批处理时，每条跨设备边独立计费：
      \[
        T_{edge}=L+\frac{s}{B}
      \]
    \item 开启 \texttt{batch\_transfers=true} 后，同一个源节点的多条跨设备出边会作为一次网络事务发送：
      \[
        T_{batch}=L+\frac{\sum_k s_k}{B}
      \]
    \item 这样既保持网络非重叠，又避免连续出边重复支付固定时延。
  \end{itemize}
\end{frame}

\begin{frame}{优化目标与可行性约束}
  求解器最小化加权目标：
  \[
    J =
    w_{avg} L_{avg}
    + w_{max} L_{max}
    + w_{util} L_{util}
  \]
  其中
  \[
    L_{avg}=\frac{\tau}{\max(scale_{avg}, 1)},\quad
    L_{max}=\frac{\tau_{frame}^{max}}{\max(scale_{max}, 1)},\quad
    L_{util}=1-\frac{device\_active\_time}{pipeline\_work\_span}
  \]
  时延归一化尺度来自全端侧/全边侧基线。
  \vspace{0.5em}
  \begin{block}{时延上限}
    正数上限是硬约束：
    \[
      \tau \leq latency\_limit,\quad
      \tau_{frame}^{max} \leq max\_frame\_latency\_limit
    \]
    取值为 0 表示关闭对应约束；超过任一启用上限的候选方案会在目标函数比较前被直接拒绝。
  \end{block}
\end{frame}

\begin{frame}{求解模式}
  \begin{table}
    \centering
    \small
    \begin{tabular}{p{0.2\textwidth}p{0.23\textwidth}p{0.46\textwidth}}
      \toprule
      模式 & 适用场景 & 行为 \\
      \midrule
      Auto & 默认模式 & 自由节点 $\leq 12$ 时枚举，否则随机搜索 \\
      Enumerate & 小规模 DAG & 遍历所有方案，返回全局最优可行解 \\
      Random Search & 较大 DAG & 随机扰动放置方案，保留最佳可行候选 \\
      Simulated Annealing & 跳出局部最优 & 按温度接受部分更差移动 \\
      \bottomrule
    \end{tabular}
  \end{table}
  \vspace{0.5em}
  所有模式都会强制固定端侧节点，并执行平均端到端和最大单帧时延上限过滤。
\end{frame}

\begin{frame}{可视化设计}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \textbf{拓扑 SVG}
      \begin{itemize}
        \item 配置 DAG 在求解前展示数据量以及本地/跨设备边样式。
        \item 求解后的展开流水线使用高对比填充色表示帧编号。
        \item 蓝色轮廓表示端侧，绿色轮廓表示边侧。
        \item 加粗或虚线源节点样式表示固定端侧和输入源事件。
        \item FIFO、传输、本地依赖和释放线都跟随源帧的同一颜色。
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{时序 SVG}
      \begin{itemize}
        \item Input、Host、Transfer、Device 四条泳道。
        \item 算子条由开始/完成时间决定，并使用与拓扑图一致的帧填充色。
        \item 传输条来自解析器的真实传输记录，可展示串行传输和批处理传输。
        \item 输入释放点、周期跨度和垂直释放/依赖线都使用对应帧颜色。
        \item 使用浏览器原生滚动和缩放控件。
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{当前技术栈}
  \begin{itemize}
    \item Python 3.9+。
    \item PyQt6 负责桌面控件和布局。
    \item PyQt6-WebEngine 负责 SVG 拓扑图和时序图渲染。

    \item 启动命令：\texttt{python -m coopinfer} 或 \texttt{python -m coopinfer.app}。

\end{frame}
